% Minimal — a two-column CV with a dark info sidebar and a header band.
%
% The layout is drawn with eso-pic: a full-height dark LEFT column on every
% page, a light header band on page 1, and the content column on white.
%
% The slot markers below are replaced with the profile's content. Everything
% outside them is ordinary LaTeX that you can edit freely.
%
% Build:  tectonic cv.tex

% Line breaking in the right column. This build has NO English hyphenation
% pattern files, and fontspec's default HyphenChar=-1 disables the hyphen

  }%
}

% dark left column: the LEFT column plus the gutter, full page height, every

% stretch.
\newcommand{\leftcolW}{254.8pt}
\newcommand{\washW}{295pt}
\newcommand{\drawDarkCol}{%
  \AddToShipoutPictureBG{%
    \AtPageUpperLeft{\mbox{\raisebox{-\paperheight}{%

  }%
}

% ---- shared macros, default = RIGHT (dark-ink) colors ----
\newcommand{\faIco}[1]{\makebox[1.15em][l]{\color{IcoCol}#1}\hspace{0.5em}}

% ---- section rhythm (tunable) --------------------------------------------
% \secspace  air ABOVE a section heading.
% \rulegap   air between the heading and its gold rule.
% \headgap   air between the rule and the section's first line.
\newcommand{\secspace}{12pt}
\newcommand{\rulegap}{4pt}
\newcommand{\headgap}{3pt}
\newcommand{\secheadrule}[2]{%
  \nobreak\vspace{\rulegap}\nobreak\nointerlineskip
  {\color{#2}\rule{#1}{0.6pt}}\par\nobreak\vspace{\headgap}}
\newcommand{\sechead}[1]{%
  \par\vspace{\secspace}%
  {\large\color{Ink}\rmfamily\bfseries #1\par}%
  \secheadrule{\linewidth}{RuleGold}}
\newcommand{\secheadfirst}[1]{%
  {\large\color{Ink}\rmfamily\bfseries #1\par}%
  \secheadrule{\linewidth}{RuleGold}}
\newcommand{\jobhead}[1]{\par\vspace{\jobspace}{\color{Ink}\rmfamily\bfseries #1}\par\nobreak\vspace{\titlegap}}
\newcommand{\jobheadfirst}[1]{{\color{Ink}\rmfamily\bfseries #1}\par\nobreak\vspace{\titlegap}}
\newcommand{\jobdates}[1]{{\small\color{InkSoft} #1}\par\nobreak}

% ---- job/project block rhythm (right column) -----------------------------
% Hierarchy rule: \jobspace > \bulletspace > \itemspace, and \titlegap
% visibly separates the title from its own details.
\newcommand{\jobspace}{8pt}
\newcommand{\titlegap}{3pt}
\newcommand{\bulletspace}{4pt}
\newcommand{\itemspace}{1pt}

% One environment for every bullet list under a job or project title, so the
% indent + rhythm are tuned in ONE place.
\newenvironment{jobitems}{%
  \par\vspace{\bulletspace}%
  \begin{itemize}[leftmargin=12pt,labelwidth=10pt,labelsep=4pt,%
                   itemsep=\itemspace,topsep=0pt,parsep=0pt]}{%
  \end{itemize}}
